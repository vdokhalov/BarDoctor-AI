function bdCurrentExpenseRows(report,result) {
  const known=value=>typeof value==='number'&&Number.isFinite(value);
  const rows=[{key:'cost',label:'Себестоимость продаж',amount:known(report.costOfGoods)?report.costOfGoods:null},{key:'salary',label:'Зарплаты',amount:known(result.payroll)?result.payroll:null}];
  const operating=report.operatingExpenses,writeoffs=report.writeoffs;
  // Split only a known contained component; never add purchases or count writeoffs twice.
  if(known(operating)&&known(writeoffs)&&writeoffs>=0&&writeoffs<=operating){
    if(writeoffs!==0)rows.push({key:'writeoffs',label:'Списания',amount:writeoffs});
    const rest=Math.round((operating-writeoffs)*100)/100;if(rest!==0)rows.push({key:'other',label:'Другие расходы',amount:rest});
  }else if(!known(operating)||operating!==0)rows.push({key:'other',label:'Другие расходы',amount:known(operating)?operating:null});
  const adjustment=report.inventoryAdjustmentNet;
  if(!known(adjustment)||adjustment!==0)rows.push({key:'stock',label:known(adjustment)?adjustment>0?'Излишки склада':'Недостачи склада':'Корректировки склада',amount:known(adjustment)?-adjustment:null});
  return rows;
}
