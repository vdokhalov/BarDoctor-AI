function bdMonthlyRatioV165(value, total) {
  const numerator = Number(value);
  const denominator = Number(total);
  if (!Number.isFinite(numerator) || !Number.isFinite(denominator) || denominator <= 0) return null;
  return Math.round((numerator / denominator) * 1000) / 10;
}

function bdMonthlyRatioLabelV165(value) {
  return value === null ? "—" : String(value).replace(".", ",") + "%";
}

function bdMonthlyChangeV165(current, previous, mode) {
  const currentValue = Number(current);
  const previousValue = Number(previous);
  if (!Number.isFinite(currentValue) || !Number.isFinite(previousValue)) return null;
  const delta = currentValue - previousValue;
  const percent = previousValue === 0 ? null : Math.round((delta / Math.abs(previousValue)) * 1000) / 10;
  const sign = delta > 0 ? "+" : delta < 0 ? "−" : "";
  const absolute = Math.abs(delta);
  const value = mode === "ratio"
    ? sign + String(Math.round(absolute * 10) / 10).replace(".", ",") + " п.п."
    : sign + (sign ? " " : "") + bdMoney2(absolute);
  return {
    delta,
    percent,
    value: value + (mode === "ratio" || percent === null ? "" : " · " + (percent > 0 ? "+" : percent < 0 ? "−" : "") + String(Math.abs(percent)).replace(".", ",") + "%"),
  };
}

function bdMonthlyShiftAnalysisV165({ report, period, navigate }) {
  const shifts = Array.isArray(report.shiftEstimates) ? report.shiftEstimates : [];
  if (!shifts.length) {
    return i.jsxs("section", { className: "bd-monthly-card-v165 bd-monthly-empty-v165", children: [
      i.jsx(bdMonthlySectionHeaderV165, { title: "Результат по сменам", caption: "Операционная картина месяца" }),
      i.jsx("p", { children: "Нет завершённых смен с финансовыми данными." }),
    ] });
  }
  const resultOf = (shift) => shift.estimatedResult ?? shift.resultBeforeCost;
  const profitable = shifts.filter((shift) => Number(resultOf(shift)) >= 0).length;
  const losing = shifts.length - profitable;
  const average = shifts.reduce((sum, shift) => sum + Number(resultOf(shift) || 0), 0) / shifts.length;
  const ordered = [...shifts].sort((left, right) => Number(resultOf(right) || 0) - Number(resultOf(left) || 0));
  const best = ordered[0];
  const worst = ordered[ordered.length - 1];
  return i.jsxs("section", { className: "bd-monthly-card-v165 bd-monthly-shifts-v165", children: [
    i.jsx(bdMonthlySectionHeaderV165, { title: "Операционный результат по сменам", caption: report.costOfGoods !== null ? "Себестоимость распределена пропорционально выручке" : "Показан результат без себестоимости", action: () => navigate("/finance?month=" + period + "&view=revenue"), actionLabel: "Открыть смены" }),
    i.jsx("div", { className: "bd-monthly-shift-summary-v165", children: [
      { key: "all", label: "Завершено", value: shifts.length },
      { key: "profit", label: "Прибыльных", value: profitable },
      { key: "loss", label: "Убыточных", value: losing },
      { key: "average", label: "Средний результат", value: bdMoney2(average) },
      { key: "best", label: "Лучшая", value: sg(best.date) + " · " + bdMoney2(resultOf(best)) },
      { key: "worst", label: "Худшая", value: sg(worst.date) + " · " + bdMoney2(resultOf(worst)) },
    ].map((entry) => i.jsxs("article", { className: "is-" + entry.key, children: [i.jsx("small", { children: entry.label }), i.jsx("strong", { children: entry.value })] }, entry.key)) }),
    i.jsx("div", { className: "bd-monthly-shift-list-v165", children: shifts.map((shift) => {
      const result = resultOf(shift);
      return i.jsxs("button", { type: "button", onClick: () => navigate("/finance?month=" + period + "&view=revenue"), children: [
        i.jsxs("span", { children: [i.jsx("strong", { children: sg(shift.date) }), i.jsxs("small", { children: ["Выручка ", bdMoney2(shift.revenue), " · Себестоимость ", shift.estimatedCost === null ? function bdMonthlyReadinessV165({ report, navigate }) {
  if (report.isClosed) return null;
  const issues = [];
  if (report.openingInventory === null) issues.push("не внесены начальные остатки");
  if (report.closingInventory === null) issues.push("не внесены конечные остатки");
  if (report.inventoryMismatch) issues.push("есть расхождение остатков");
  if (report.coveragePercent < 100) issues.push("не объяснены " + Math.max(0, report.expectedShifts - report.accountedShifts) + " смен");
  if (!report.periodPast) issues.push("месяц ещё не завершён");
  return i.jsxs("section", { className: "bd-monthly-card-v165 bd-monthly-readiness-v165", children: [
    i.jsx(bdMonthlySectionHeaderV165, { title: "Готовность отчёта", caption: "Что мешает финальному закрытию" }),
    i.jsxs("div", { className: "bd-monthly-readiness-score-v165", children: [
      i.jsx("strong", { children: report.coveragePercent + "%" }),
      i.jsx("div", { children: i.jsx("i", { style: { width: Math.max(0, Math.min(100, report.coveragePercent)) + "%" } }) }),
    ] }),
    issues.length ? i.jsx("ul", { children: issues.map((issue) => i.jsx("li", { children: issue }, issue)) }) : i.jsx("p", { children: "Данные прошедших смен учтены. Можно продолжить закрытие месяца." }),
    i.jsx("button", { type: "button", className: "bd-monthly-primary-action-v165", onClick: () => navigate("/reports?closeMonth=1"), children: "Продолжить закрытие" }),
  ] });
}

function bdMonthlyCurrencyBoundaryNoticeV320({ report }) {
  const totals = (report?.excludedForeignCurrencyTotals || []).filter((row) => Number(row.amount) !== 0);
  if (!totals.length) return null;
  const labels = { purchase: "Закупки", expense: "Платежи/расходы", revenue: "Выручка", snapshot: "Остатки" };
  return i.jsxs("section", { "data-bd-currency-boundary": "v320", className: "bd-monthly-currency-boundary-v320", role: "status", children: [
    i.jsx("strong", { children: "Иностранная валюта не включена в " + report.accountingCurrency }),
    i.jsx("p", { children: totals.map((row) => (labels[row.kind] || row.kind) + ": " + new Intl.NumberFormat("ru-RU", { maximumFractionDigits: 2 }).format(Number(row.amount)) + " " + row.currency).join(" · ") + " — нет сохранённого исторического курса." }),
  ] });
}
