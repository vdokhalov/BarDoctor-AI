function bdMonthlyRatioV165(value, total) {
  if (typeof value !== "number" || typeof total !== "number") return null;
  const numerator = Number(value);
  const denominator = Number(total);
  if (!Number.isFinite(numerator) || !Number.isFinite(denominator) || denominator <= 0) return null;
  return Math.round((numerator / denominator) * 1000) / 10;
}

function bdMonthlyRatioLabelV165(value) {
  return value === null ? "—" : String(value).replace(".", ",") + "%";
}

function bdMonthlyChangeV165(current, previous, mode) {
  const currentValue = Number(current);
  const previousValue = Number(previous);
  if (!Number.isFinite(currentValue) || !Number.isFinite(previousValue)) return null;
  const delta = currentValue - previousValue;
  const percent = previousValue === 0 ? null : Math.round((delta / Math.abs(previousValue)) * 1000) / 10;
  const sign = delta > 0 ? "+" : delta < 0 ? "−" : "";
  const absolute = Math.abs(delta);
  const value = mode === "ratio"
    ? sign + String(Math.round(absolute * 10) / 10).replace(".", ",") + " п.п."
    : sign + (sign ? " " : "") + bdMoney2(absolute);
  return {
    delta,
    percent,
    value: value + (mode === "ratio" || percent === null ? "" : " · " + (percent > 0 ? "+" : percent < 0 ? "−" : "") + String(Math.abs(percent)).replace(".", ",") + "%"),
  };
}

// Presentation only: never recompute COGS or turn a current result into a closed-month profit.
function bdCurrentResultPresentation(report) {
  const known=value=>typeof value==="number"&&Number.isFinite(value);
  const estimated=report.currentFinance&&!known(report.resultBeforeCost)&&known(report.provisionalResultBeforeCost);
  const before=estimated?report.provisionalResultBeforeCost:report.resultBeforeCost;
  const after=estimated?report.provisionalResultAfterCost:report.resultAfterCost;
  const afterCost=report.currentFinance&&known(report.costOfGoods)&&known(after);
  return {afterCost,estimated,costKnown:known(report.costOfGoods),payroll:estimated?report.provisionalPayroll:report.payroll,
    value:report.currentFinance?(afterCost?after:known(before)?before:null):known(report.operatingResult)?report.operatingResult:null,
    label:report.currentFinance?(afterCost?(after<0?"Убыток за месяц":"Прибыль за месяц"):known(before)?"Результат до себестоимости":"Прибыль за месяц"):report.isClosed?"Чистая прибыль месяца":"Предварительный результат месяца",
    explanation:estimated?report.provisionalExplanation:afterCost?"Выручка за вычетом учтённых расходов, начисленного ФОТ и подтверждённой себестоимости, с учётом складских корректировок. Это текущий результат по внесённым данным; месяц ещё не закрыт.":report.resultExplanation
  };
}

function bdMonthlyReadControllerV513(load, publish) {
  let disposed = false, paused = false, sequence = 0, controller;
  return {
    async refresh() {
      if (disposed || paused) return;
      const current = ++sequence;
      controller?.abort(); controller = new AbortController();
      try { const value = await load(controller.signal); if (!disposed && current === sequence) publish({ ...value, error: null }); }
      catch (error) { if (!disposed && current === sequence) publish({ report: null, reasons: [], error: error.message }); }
    },
    pause() { paused = true; ++sequence; controller?.abort(); },
    resume() { if (!disposed) paused = false; },
    dispose() { disposed = true; ++sequence; controller?.abort(); },
  };
}

function bdUseMonthlyReportV513(venueId, monthKey, revision) {
  const key = venueId + ":" + monthKey;
  const [state, setState] = S.useState(null);
  const [retry, setRetry] = S.useState(0);
  S.useEffect(() => {
    if (!Number.isSafeInteger(venueId)) return;
    const reader = bdMonthlyReadControllerV513(signal => bdFetchMonthlyReportV513(venueId, monthKey, signal), value => setState({ key, ...value }));
    reader.refresh();
    const refresh = () => { if (document.visibilityState === "visible") reader.refresh(); };
    const pause = () => reader.pause();
    const resume = () => { reader.resume(); refresh(); };
    const timer = setInterval(refresh, 15000);
    window.addEventListener("focus", refresh); document.addEventListener("visibilitychange", refresh);
    window.addEventListener("pagehide", pause); window.addEventListener("pageshow", resume);
    return () => { reader.dispose(); clearInterval(timer); window.removeEventListener("focus", refresh); document.removeEventListener("visibilitychange", refresh); window.removeEventListener("pagehide", pause); window.removeEventListener("pageshow", resume); };
  }, [key, revision, retry]);
  return { ...(state?.key === key ? state : { report: null, error: null, reasons: [] }), retry: () => setRetry(value => value + 1) };
}

  try { url = new URL(path, "https://bardoctor.invalid"); } catch { return null; }
  if (url.origin !== "https://bardoctor.invalid" || !allowed.includes(url.pathname) || !/^\d{4}-(0[1-9]|1[0-2])$/.test(monthKey) || !Number.isSafeInteger(venueId) || venueId <= 0) return null;

  url.searchParams.set("month", monthKey); url.searchParams.set("venue", String(venueId)); url.hash = "";

}
