function bdRecipeProductV511(line, products, venueId) {
  if(line.linkStatus==="missing"||line.resolutionStatus==="reference_conflict"||line.purchaseProductKey&&line.productKey&&line.purchaseProductKey!==line.productKey)return undefined;
  const key=line.purchaseProductKey||line.productKey, id=line.nomenclatureItemId;
  return products.find(p=>(p.key||p.productKey)===key&&(!id||String(p.id||p.nomenclatureItemId||p.key)===String(id))&&p.active!==false&&!p.archived&&p.status!=="archived"&&p.kind!=="service"&&p.type!=="service"&&(!p.venueId||Number(p.venueId)===Number(venueId)));
}
function bdRecipeAmountV511(line, product) {
  if(["requires_review","invalid","missing"].includes(line.unitResolutionStatus))return null;
  const amount=bdCatToBase(line.quantity,line.unit), target=bdCatToBase(1,product?.unit||product?.baseUnit);
  if(!(amount.amount>0)||amount.unit==="unknown"||target.unit==="unknown")return null;
  if(amount.unit===target.unit)return amount;
  const conversion=line.unitConversion;
  if(amount.unit==="pcs"&&conversion?.confirmedByUser===true&&Number(conversion.amount)>0){
    const converted=bdCatToBase(Number(conversion.amount),conversion.unit);
    if(converted.amount>0&&Number.isFinite(converted.amount)&&converted.unit===target.unit)return {amount:amount.amount*converted.amount,unit:converted.unit};
  }
  return null;
}

function bdRecipeLineCostV511(line, costs, products, venueId) {
  const product=bdRecipeProductV511(line,products,venueId),amount=product&&bdRecipeAmountV511(line,product);
  const unknown=reason=>({id:line.id,complete:false,cost:null,unitPrice:null,reason,costStatus:"UNKNOWN"});
  if(!product)return unknown("mapping");
  if(!amount)return unknown("unit");
  const key=product.key||product.productKey,id=String(product.id||product.nomenclatureItemId||key);
  // A stable line ID alone cannot authorize a price after changing its stock product.
  const matches=costs.filter(c=>{
    const refs=[c.productKey,c.purchaseProductKey].filter(Boolean);
    return (refs.length?refs.every(ref=>ref===key):String(c.nomenclatureItemId||"")===id)&&(!c.nomenclatureItemId||String(c.nomenclatureItemId)===id)&&(!c.venueId||Number(c.venueId)===Number(venueId));
  });
  const cost=matches.find(c=>!c.id)||matches.find(c=>String(c.id)===String(line.id))||matches[0];
  const price=typeof cost?.unitPrice==="number"||typeof cost?.unitPrice==="string"&&cost.unitPrice.trim()?Number(cost.unitPrice):NaN;
  if(cost?.complete!==true||!Number.isFinite(price)||price<0||!cost.currency||cost.costStatus==="UNKNOWN")return unknown("price");
  const basis=bdTechCostUnitV376(cost.unit);
  if(basis.unit!==amount.unit||!(basis.factor>0))return unknown("price_unit");
  const quantity=amount.amount/basis.factor;
  if(!Number.isFinite(quantity*price))return unknown("unit");
  return {...cost,id:line.id,complete:true,reason:null,amount:quantity,cost:Math.round(quantity*price*100)/100,unitPrice:price};
}
