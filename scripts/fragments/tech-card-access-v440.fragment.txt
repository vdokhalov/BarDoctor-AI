function bdUseRecipeRefreshV442(recipe,balances,dirty,saving,setRecipe,setBalances) {
  const token=JSON.stringify(recipe||null);
  const previous=S.useRef(token);
  const [blocked,setBlocked]=S.useState(false);
  const changed=previous.current!==token;
  S.useEffect(()=>{
    if(previous.current===token||saving.current)return;
    previous.current=token;
    if(dirty.current||!recipe){setBlocked(true);return;}
    setRecipe({...recipe,ingredients:bdCatArray(recipe.ingredients).map(line=>({...line}))});
    const next={};
    for(const line of bdCatArray(recipe.ingredients)){
      const key=bdCatBalanceKey(line),balance=(balances||[]).find(row=>row.key===key)||{};
      next[line.id]={key,current:balance.current??'',safety:balance.safety??'',onOrder:balance.onOrder??'',packageAmount:balance.packageAmount??'',checkedAt:balance.checkedAt||''};
    }
    setBalances(next);
    setBlocked(false);
  },[token,recipe,balances,dirty,saving,setRecipe,setBalances]);
  return blocked||(changed&&dirty.current&&!saving.current);
}
